13PALETTE = (

27def color_for(detection: Detection) -> tuple[int, int, int]: 

28 rank = detection.rank or 1 

29 return PALETTE[(rank - 1) % len(PALETTE)]

32def draw_detections(frame: np.ndarray, detections: Sequence[Detection]) -> np.ndarray: 

33 """Return a copy of ``frame`` with each detection's box and label drawn on it.""" 

34 out = frame.copy() 

35 for detection in detections: 

36 _draw(out, detection)

40def _draw(out: np.ndarray, detection: Detection) -> None: 

41 height, width = out.shape[:2] 

42 px = detection.box.to_pixels(width, height) 

43 color = color_for(detection) 

44 cv2.rectangle(out, (px.x1, px.y1), (px.x2, px.y2), color, THICKNESS) 

45 

46 label = detection.label() 

47 (tw, th), baseline = cv2.getTextSize(label, FONT, FONT_SCALE, 2) 

48 # Put the label above the box, or inside it when the box touches the top edge. 

49 y_top = px.y1 - th - baseline - 6 if px.y1 >= th + baseline + 6 else px.y1 

50 cv2.rectangle(out, (px.x1, y_top), (px.x1 + tw + 10, y_top + th + baseline + 6), color, cv2.FILLED) 

51 cv2.putText(out, label, (px.x1 + 5, y_top + th + 3), FONT, FONT_SCALE, TEXT_COLOR, 2, cv2.LINE_AA)
